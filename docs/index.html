bash — Pull Docker Image & Ingest Transaction

# Pull and run pre-built container from GitHub Container Registry (GHCR)

docker pull ghcr.io/crnjihia/spend-tracker-api:latest

docker run -d -p 8000:8000 --env-file .env ghcr.io/crnjihia/spend-tracker-api:latest


# Ingest an M-Pesa C2B Paybill payment payload

curl -X POST http://127.0.0.1:8000/api/v1/transactions/mpesa \

-H "Authorization: Bearer $ACCESS_TOKEN" \

-H "Content-Type: application/json" \

-d '{"TransID":"QGH7XYZ123","TransAmount":"1500.00","BusinessShortCode":"247247","BillRefNumber":"Naivas Supermarket","TransTime":"20261008143022"}'
